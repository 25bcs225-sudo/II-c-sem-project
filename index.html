class Parent { public int publicValue = 10; protected int protectedValue = 20; int defaultValue = 30; private int privateValue = 40; public void displayPrivateValue() { System.out.println("Private Value: " + privateValue); } } class Child extends Parent { public void displayValues() { System.out.println("Public Value: " + publicValue); System.out.println("Protected Value: " + protectedValue); System.out.println("Default Value: " + defaultValue); // Private members cannot be accessed directly // System.out.println(privateValue); // Access private data through a public method displayPrivateValue(); } } public class Main { public static void main(String[] args) { Child obj = new Child(); obj.displayValues(); } }
